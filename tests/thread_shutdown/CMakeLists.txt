cmake_minimum_required(VERSION 3.22)
project(citron_thread_shutdown_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(THREAD_SHUTDOWN_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(Qt6 REQUIRED COMPONENTS Core)
enable_testing()
add_executable(thread-shutdown thread-shutdown.cpp)
target_include_directories(thread-shutdown PRIVATE ../../src)
target_link_libraries(thread-shutdown PRIVATE Qt6::Core)
if(MSVC)
    target_compile_options(thread-shutdown PRIVATE /UNDEBUG)
else()
    target_compile_options(thread-shutdown PRIVATE -UNDEBUG)
endif()
if(THREAD_SHUTDOWN_SANITIZERS)
    target_compile_options(thread-shutdown PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(thread-shutdown PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME thread-shutdown COMMAND thread-shutdown)
set_tests_properties(thread-shutdown PROPERTIES TIMEOUT 10)
